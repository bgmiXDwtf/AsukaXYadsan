--------------------------------------------------
-- ✅ PERFECT DRAG SYSTEM (PC + MOBILE)
--------------------------------------------------

local dragging = false
local dragStart = nil
local startPos = nil

local function updateDrag(input)
	local delta = input.Position - dragStart
	frame.Position = UDim2.new(
		startPos.X.Scale,
		startPos.X.Offset + delta.X,
		startPos.Y.Scale,
		startPos.Y.Offset + delta.Y
	)
end

title.InputBegan:Connect(function(input)
	if minimized then return end

	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = frame.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(input)
	if minimized then return end

	if dragging and (
		input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch
	) then
		updateDrag(input)
	end
end)

--------------------------------------------------
-- TELEPORT LOOP
--------------------------------------------------

local function startTeleport()
	if runningLoop then return end

	character, hrp = getCharacter()
	originalCFrame = hrp.CFrame
	runningLoop = true

	task.spawn(function()
		local angle = 0

		while runningLoop do
			local radius = tonumber(distanceBox.Text) or 20
			local speed = tonumber(speedBox.Text) or 0.2

			angle += math.rad(25)

			local x = math.cos(angle) * radius
			local z = math.sin(angle) * radius

			if hrp and originalCFrame then
				hrp.CFrame = originalCFrame + Vector3.new(x, 0, z)
			end

			task.wait(speed)
		end
	end)
end

local function stopTeleport()
	runningLoop = false

	if hrp and originalCFrame then
		hrp.CFrame = originalCFrame
	end
end

--------------------------------------------------
-- BUTTON CONNECTIONS
--------------------------------------------------

startBtn.MouseButton1Click:Connect(startTeleport)
stopBtn.MouseButton1Click:Connect(stopTeleport)

--------------------------------------------------
-- AUTO RESET IF PLAYER DIES / RESPAWNS
--------------------------------------------------

player.CharacterAdded:Connect(function()
	task.wait(1)
	character, hrp = getCharacter()

	if runningLoop then
		originalCFrame = hrp.CFrame
	end
end)
